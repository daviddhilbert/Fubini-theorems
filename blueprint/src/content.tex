\begin{definition}[Presentations in degree at most k]
\label{weak-objects}
\lean{DiamondIntegral.Section5.NativeWeakRVObject, DiamondIntegral.Section5.nativeWeakRVObjectLift}
\leanok
\uses{ground}
A presentation in degree at most $k$ over a parameter base is a definable RV set $U$ with a definable finite-to-one label $p:U\to\mathrm{RV}_\infty^k$; labels may vanish. Its lift is $\{(x,u):\rv(x)=p(u)\}$, and its zero VF coordinates are retained. For $I\subseteq\{1,\ldots,k\}$, the exact stratum on which precisely the labels in $I$ are nonzero is an object of $\mathrm{RV}[|I|]$. This is the paper's $\mathrm{RV}[*]$ convention.
\end{definition}

\begin{proof}
\uses{ground}
The structure stores the arity, the domain, the label map and finiteness of its fibres. The lift is a ground-definable set with finite RV fibres.
\end{proof}

\begin{definition}[Values of a presentation]
\label{weak-values}
\lean{DiamondIntegral.Section5.nativeWeakGeometricValue, DiamondIntegral.Section5.nativeWeakGeometricValue_eq_sum_strata, DiamondIntegral.Section5.nativeVFGeometricValue_weak_lift}
\leanok
\uses{weak-objects, integral}
The bounded value of a presentation is the bounded value of its underlying RV set, the sum of the values of its exact strata. Its geometric value is $\sum_I L^{-|I|}\widetilde{\mathbb E}_g(U_I)$: each exact stratum is normalized in its own degree. Over the original ground both are the values of the integral of the lift, and on single-degree objects they are the usual values.
\end{definition}

\begin{proof}
\uses{weak-objects, integral}
The lift is an RV pullback, partitioned by the zero pattern of its VF coordinates; the piece with nonzero coordinates $I$ is the lift of $U_I$ with the other coordinates zero. A single-degree object has only the full stratum.
\end{proof}

\begin{lemma}[Thickening of vanishing labels]
\label{thickening}
\lean{DiamondIntegral.Section5.thickAt, DiamondIntegral.Section5.thickFull_row_value, DiamondIntegral.Section5.thickSeq_vf}
\leanok
\uses{weak-values}
Replacing a vanishing label, on its zero stratum, by a point of $\mathbb A_{\mathrm{RES}}=\mathbb G_m\sqcup\mathrm{pt}$ with its nonzero label is a first-order operation on the original-ground formulas. After all labels are thickened, the labels are nonzero and the geometric value is unchanged. Over an original-ground parameter set, the geometric value of the integral of the lift family is unchanged as well.
\end{lemma}

\begin{proof}
\uses{weak-values}
On strata, $U_I$ becomes $U_I\times\mathbb A_{\mathrm{RES}}$ in degree $|I|+1$, and $\widetilde{\mathbb E}_g(\mathbb A_{\mathrm{RES}})=L$. On lifts, the zero stratum of the coordinate is replaced by its product with $\mathcal O^\times\sqcup(1+\mathcal M)$, whose integral has geometric value $1$.
\end{proof}

\begin{definition}[Value-group families and their inflation]
\label{gamma-family}
\lean{DiamondIntegral.Section5.GammaVFFamily, DiamondIntegral.Section5.GammaVFFamily.inflated}
\leanok
\uses{objects}
A family $\mathcal A\subseteq\Gamma^n\times\mathrm{VF}^m\times\mathrm{RV}^s$ definable over the original ground has finite RV fibres over $\Gamma^n\times\mathrm{VF}^m$. Its row $A_\gamma$ is an object over the ground when every coordinate of $\gamma$ is definable there. Its inflation is $A^{\mathrm{inf}}=\{(x,a,t):x\in(\mathrm{VF}^\times)^n,\ (v(x),a,t)\in\mathcal A\}$.
\end{definition}

\begin{proof}
\uses{objects}
Definability of the rows and of the inflation is proved through full RV code preimages of the value-group coordinates; no representative is named.
\end{proof}

\begin{lemma}[Uniform representatives over value-group parameters]
\label{uniform-reps}
\lean{DiamondIntegral.Section5.GammaVFFamily.uniform_representatives}
\leanok
\uses{gamma-family, lifting, weak-values}
There are finitely many original-ground RV families $\mathcal U_l\subseteq\Gamma^n\times\mathrm{RV}^{r_l}$ over a definable partition $(P_l)$ of $\Gamma^n$ with three properties. A row of $\mathcal A$ is empty exactly when all rows of the $\mathcal U_l$ are. At a definable $\gamma\in P_l$, the bounded value of $A_\gamma$ is that of the row of $\mathcal U_l$. The bounded value of $A^{\mathrm{inf}}$ is the sum of those of the RV inflations of the $\mathcal U_l$.
\end{lemma}

\begin{proof}
\uses{gamma-family, lifting, weak-values}
At each $\gamma$, choose a lift over the ground with $\gamma$ adjoined, promote its formulas to value-group families, and use compactness for the first-order validity of a datum. Actual bijections give both value identities.
\end{proof}

\begin{lemma}[Relative monomial charts]
\label{relative-chart}
\lean{DiamondIntegral.Section5.inflated_relative_block_chart, DiamondIntegral.Section5.relative_monomial_decomposition}
\leanok

Let an original-ground $X\subseteq(\mathrm{RV}^\times)^n\times(\mathrm{RV}^\times)^r$ depend on $z$ only through $\operatorname{vrv}(z)$. Then $X$ has a finite definable partition into charts. \end{lemma}

\begin{proof}
\uses{monomial}
Apply the monomial decomposition, reduce the $z$ block by its saturated left kernel, use divisibility of residue units, and extend the right-invertible rows to $N$. The value set is the value image of the piece.
\end{proof}

\begin{proof}
\uses{relative-chart}
The coordinate change maps each chart onto $W_i\times J_i^\sharp$ after a permutation; apply the monomial evaluation of the bounded retraction.
\end{proof}

\begin{lemma}[A separate base partition]
\label{base-partition}
\lean{DiamondIntegral.Section5.inflated_gamma_base_partition}
\leanok

For finitely many definable $J_i\subseteq\Gamma^n\times\Gamma^{\ell_i}$ and definable $D_k\subseteq\Gamma^n$ there is a finite definable partition $(E_a)$ of $\Gamma^n$ refining the $D_k$, on which every slice Euler characteristic $\chi_b(J_{i,\gamma})$ is a constant $e_{ia}$, zero for empty slices. \end{proof}

Over $\mathbb C((t))$, with $\mathrm{Vol}=\Theta\circ\mathbb E_b\circ\int$, the same partition satisfies $\mathrm{Vol}(A_\gamma)=\Theta(v_a)$ at every $\gamma\in E_a$ with ground-definable, that is rational, coordinates, and $\mathrm{Vol}(A^{\mathrm{inf}})=[\mathbb G_m]^n\sum_a\chi_b(E_a)\Theta(v_a)$. The statement is instantiated in the constructed ambient.
\end{theorem}

\begin{proof}
\uses{inflated, theta}
Apply the ring homomorphism $\Theta$ to the inflated-family theorem and use $\Theta(g)=[\mathbb G_m]$.
\end{proof}
