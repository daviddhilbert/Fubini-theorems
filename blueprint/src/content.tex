\begin{lemma}[Relative Gamma push over an RV base]
\label{relative-gamma}
\lean{DiamondIntegral.Section5.native_cylinder_fubini, DiamondIntegral.Section5.native_relative_gamma_fubini_partition}
\leanok
\uses{monomial, euler}
For an RV base $X$, a definable family $J_x\subseteq\Gamma^r$, and its full RV lift $Y$, the bounded value is $\mathbb E_b[Y]=g^r\sum_e e\mathbb E_b[X_e]$, where $X_e=\{x:\chi_b(J_x)=e\}$ and $g=[\mathbf k^\times]$. The sum and the level-set partition are finite.
\end{lemma}

\begin{proof}
\uses{monomial, euler}
Prepare unary endpoint graphs by a finite monomial catalogue and calculate their cylinder factors. Retain the RV base and lift the Gamma coordinates in reverse order. Induction on each cylindrical tree gives its shape factor times $g^r$; finite certificate partitions and additivity give the formula. Singleton fibres yield the varying value-torsor identity.
\end{proof}

\begin{theorem}[Theorem 5.17: bounded Fubini]
\label{main}
\lean{DiamondIntegral.Section5.bounded_fubini_discharged}
\leanok
\uses{ground, linear, lifting, integral, fibre, native}
Let $A,A'$ be finite RV-fibre VF objects over the original ground and let $f:A\to C$, $f':A'\to C$ be ground-definable maps to a finite mixed VF/RV base. If at every $z\in C$ the two fibres have parameter-definable RV lifts with equal bounded RES evaluations, then $I_b([A])=I_b([A'])$. All points of the saturated target are tested, including generic points.
\end{theorem}

\begin{proof}
\uses{ground, linear, lifting, integral, fibre, native}
Apply the native Fubini theorem with the proved value-group preparation theorem and completed original-ground lifting record. This is the only selected Palomar result. Geometric Fubini, Puiseux comparisons and the variety corollary remain in the library but are not separate challenge claims.
\end{proof}
